function mostrarInicio() { document.getElementById("inicio").style.display = "block"; document.querySelectorAll(".secao").forEach(function (secao) { secao.classList.remove("ativa"); }); } function abrirArea(nome) { document.getElementById("inicio").style.display = "none"; document.querySelectorAll(".secao").forEach(function (secao) { secao.classList.remove("ativa"); }); document.getElementById("area-" + nome).classList.add("ativa"); } document.querySelectorAll("[data-area]").forEach(function (botao) { botao.addEventListener("click", function () { abrirArea(botao.dataset.area); }); }); document.querySelectorAll(".botao-voltar").forEach(function (botao) { botao.addEventListener("click", mostrarInicio); }); document.getElementById("loginButton").addEventListener("click", async function () { const email = document.getElementById("emailInput").value.trim().toLowerCase(); if (email !== adminEmail) { message.textContent = "Este e-mail não tem acesso."; return; } try { await sendSignInLinkToEmail(auth, email, actionCodeSettings); window.localStorage.setItem("emailParaLogin", email); message.textContent = "Link enviado! Abra o e-mail e toque no link."; } catch (erro) { message.textContent = "Erro ao enviar o link: " + erro.message; } }); document.getElementById("sairButton").addEventListener("click", function () { signOut(auth); }); onAuthStateChanged(auth, function (usuario) { if (usuario && usuario.email === adminEmail) { mostrarPainel(); } else { loginArea.style.display = "block"; painel.style.display = "none"; } }); async function terminarLogin() { if (!isSignInWithEmailLink(auth, window.location.href)) return; let email = window.localStorage.getItem("emailParaLogin"); if (!email) { email = window.prompt("Confirme o e-mail do administrador:"); } try { await signInWithEmailLink(auth, email, window.location.href); window.localStorage.removeItem("emailParaLogin"); window.history.replaceState({}, document.title, window.location.pathname); } catch (erro) { message.textContent = "Erro ao entrar: " + erro.message; } } terminarLogin();
